\node[title,text width=148mm] (definition) at (0,0) {\introhead{A question about two rows becomes a join}\\[6pt]
  A semantic join tests a natural-language condition with a model.\\
  It keeps the pairs for which the model returns True.\\
  Here the condition is: \textbf{``Do these records describe the same person?''}};

\node[data,text width=65mm,anchor=north] (a) at ([xshift=-40mm,yshift=-6mm]definition.south) {\textbf{Record $a$ from table $A$}\\[6pt]
  \begin{tabular}{@{}ll@{}}
  Name & Amina Patel\\[3pt]
  Date of birth & 1988-04-12\\[3pt]
  Postcode & 3000
  \end{tabular}};
\node[data,text width=65mm,anchor=north] (b) at ([xshift=40mm,yshift=-6mm]definition.south) {\textbf{Record $b$ from table $B$}\\[6pt]
  \begin{tabular}{@{}ll@{}}
  Name & Amina Patel\\[3pt]
  Date of birth & 1988-04-12\\[3pt]
  Postcode & 3000
  \end{tabular}};
\node[fit=(a)(b),inner sep=0pt] (records) {};
\node[word,text width=148mm,anchor=north] (toy) at ([yshift=-3mm]records.south) {A teaching example with constructed records and stipulated probabilities.};

\node[model,text width=65mm,anchor=north] (fshead) at ([xshift=-40mm,yshift=-7mm]toy.south) {\introhead{Fellegi--Sunter / Splink}\\[4pt]Compare fields, then combine their evidence.};
\node[llm,text width=65mm,anchor=north] (semhead) at ([xshift=40mm,yshift=-7mm]toy.south) {\introhead{Semantic join}\\[4pt]Give both rows and the condition to an LLM.};
\coordinate (bus) at ([yshift=-3mm]toy.south);
\draw[branch] (bus) -| (fshead.north);
\draw[branch] (bus) -| (semhead.north);

\node[data,text width=65mm,anchor=north] (comparisons) at ([yshift=-5mm]fshead.south) {\textbf{What do we observe?}\\[6pt]
  \begin{tabular}{@{}ll@{}}
  Name & agrees\\[2pt]
  Date of birth & agrees\\[2pt]
  Postcode & agrees
  \end{tabular}\\[6pt]
  The comparison vector is\\[3pt]
  $\boldsymbol\gamma=(\text{agree},\text{agree},\text{agree})$.};
\node[llm,text width=65mm,anchor=north] (question) at ([yshift=-5mm]semhead.south) {\textbf{What do we ask?}\\[6pt]
  ``Do these records describe the same person?''\\[7pt]
  Supply the two Amina Patel rows above.\\[5pt]
  Fix the model, prompt and parsing rules:\\
  together they define the \textbf{reference procedure}.};
\draw[flow] (fshead) -- (comparisons);
\draw[flow] (semhead) -- (question);
\node[fit=(comparisons)(question),inner sep=0pt] (rowone) {};

\node[model,text width=65mm,anchor=north] (evidence) at ([xshift=-40mm,yshift=-5mm]rowone.south) {\textbf{What does agreement tell us?}\\[6pt]
  Suppose the name agrees in\\[3pt]
  \textbf{85 of 100 same-person pairs} ($m=0.85$),\\[3pt]
  \textbf{1 of 100 different-person pairs} ($u=0.01$).\\[7pt]
  Name agreement is $m/u=\mathbf{85}$ times as likely\\
  under the same-person model.\\[5pt]
  It multiplies the prior odds by \textbf{85}.};
\node[llm,text width=65mm,anchor=north] (decision) at ([xshift=40mm,yshift=-5mm]rowone.south) {\textbf{What comes back?}\\[7pt]
  A yes/no answer (True or False) for this pair.\\[9pt]
  \begin{tabular}{@{}ll@{}}
  \textbf{If True} & keep the pair $(a,b)$\\[8pt]
  \textbf{If False} & omit the pair $(a,b)$
  \end{tabular}\\[9pt]
  This is a conditional illustration,\\
  not a recorded model decision.};
\draw[flow] (comparisons) -- (evidence);
\draw[flow] (question) -- (decision);
\node[fit=(evidence)(decision),inner sep=0pt] (rowtwo) {};

\node[model,text width=65mm,anchor=north] (score) at ([xshift=-40mm,yshift=-5mm]rowtwo.south) {\textbf{Use all three comparisons and the prior}\\[7pt]
  Evidence factors: name $85$, birth date $450$,\\
  postcode $20$.\\[5pt]
  Before comparing fields, suppose 1 in 10,000\\
  pairs is a match: $\pi=1/10{,}000$.\\[7pt]
  $\underbrace{\dfrac{1}{9999}}_{\text{prior odds}}\ \times\ \underbrace{85\times450\times20}_{\text{comparison evidence}}$\\[9pt]
  $\Longrightarrow$ \textbf{98.7\% posterior match probability}\\[6pt]
  This calculation assumes comparisons are\\
  conditionally independent within same-person\\
  pairs and within different-person pairs.\\[7pt]
  A chosen threshold/review policy determines\\
  whether the scored pair is accepted.};
\node[llm,text width=65mm,anchor=north] (relation) at ([xshift=40mm,yshift=-5mm]rowtwo.south) {\textbf{Apply the question across the two tables}\\[7pt]
  The direct reference evaluates every pair in $A\times B$.\\[7pt]
  The join returns the pairs whose answer is True:\\[7pt]
  \begin{tabular}{@{}ll@{}}
  \textbf{Left record} & \textbf{Right record}\\[3pt]
  $a$ & $b$ \quad (only if True)
  \end{tabular}\\[9pt]
  With this identity question, retained pairs are\\
  proposed identity links.\\[7pt]
  A Boolean answer does not supply a calibrated\\
  match probability.};
\draw[flow] (evidence) -- (score);
\draw[flow] (decision) -- (relation);
\node[fit=(score)(relation),inner sep=0pt] (rowthree) {};

\node[title,text width=148mm,anchor=north] (newhead) at ([yshift=-8mm]rowthree.south) {\introhead{What LOTUS and Featurized-Decomposition Join add}};
\node[box,text width=65mm,anchor=north,draw=llmline] (lotus) at ([xshift=-40mm,yshift=-5mm]newhead.south) {\textbf{LOTUS: make the question a table operation}\\[6pt]
  Write a condition, compose table operators\\
  and optimize their execution.\\[6pt]
  The reference defines the intended result.\\
  Optional optimizations reduce execution cost\\
  under statistical accuracy targets relative to it.};
\node[box,text width=65mm,anchor=north,draw=llmline] (fdj) at ([xshift=40mm,yshift=-5mm]newhead.south) {\textbf{FDJ: reuse work across many pairs}\\[6pt]
  Derive reusable features for individual records\\[3pt]
  $\downarrow$\\[3pt]
  cheap comparisons screen candidate pairs\\[3pt]
  $\downarrow$\\[3pt]
  reference-LLM refinement produces the result.};
\node[fit=(lotus)(fdj),inner sep=0pt] (contribution) {};
\node[truth,text width=145mm,anchor=north] (validation) at ([yshift=-6mm]contribution.south) {\textbf{Check the identity links against independent identity evidence.}\\[5pt]
  Agreement with a reference LLM measures execution fidelity; it does not establish who is the same person.\\[5pt]
  Another condition, such as ``Does this report mention this medicine?'', defines a different relation\\
  and needs its own evaluation.};
